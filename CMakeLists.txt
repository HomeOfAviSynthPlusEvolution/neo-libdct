cmake_minimum_required(VERSION 3.24)
project(neo_libdct VERSION 0.1.0 LANGUAGES C CXX)
option(NEO_DCT_BUILD_TESTS "Build numerical and C ABI tests" ${PROJECT_IS_TOP_LEVEL})
option(NEO_DCT_ALLOW_FMA "Allow floating-point contraction" OFF)
include(FetchContent)
if(NOT TARGET hwy)
  set(HWY_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
  set(HWY_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(HWY_ENABLE_CONTRIB OFF CACHE BOOL "" FORCE)
  set(HWY_FORCE_STATIC_LIBS ON CACHE BOOL "" FORCE)
  FetchContent_Declare(highway
    URL https://github.com/google/highway/archive/refs/tags/1.4.0.tar.gz
    URL_HASH SHA256=e72241ac9524bb653ae52ced768b508045d4438726a303f10181a38f764a453c
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(highway)
endif()
add_library(neo_dct STATIC src/api.cpp src/backends/scalar.cpp src/backends/highway.cpp)
add_library(neo::dct ALIAS neo_dct)
target_compile_features(neo_dct PRIVATE cxx_std_17)
target_include_directories(neo_dct PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>
  PRIVATE src)
target_link_libraries(neo_dct PRIVATE hwy)
target_compile_definitions(neo_dct PRIVATE HWY_COMPILE_ALL_ATTAINABLE)
# The static library can be embedded in a host plugin/shared module.
set_target_properties(neo_dct PROPERTIES POSITION_INDEPENDENT_CODE ON
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(MSVC)
  if(NEO_DCT_ALLOW_FMA)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
      target_compile_options(neo_dct PRIVATE /clang:-fno-fast-math /clang:-ffp-contract=fast)
    else()
      target_compile_options(neo_dct PRIVATE /fp:precise /fp:contract)
    endif()
  else()
    target_compile_options(neo_dct PRIVATE /fp:strict)
  endif()
else()
  target_compile_options(neo_dct PRIVATE -fno-fast-math)
  if(NEO_DCT_ALLOW_FMA)
    target_compile_options(neo_dct PRIVATE -ffp-contract=fast)
  else()
    target_compile_options(neo_dct PRIVATE -ffp-contract=off)
  endif()
endif()
if(NEO_DCT_BUILD_TESTS)
  enable_testing()
  find_package(Threads REQUIRED)
  add_executable(neo_dct_tests tests/test_dct.cpp)
  target_compile_features(neo_dct_tests PRIVATE cxx_std_17)
  target_compile_definitions(neo_dct_tests PRIVATE HWY_COMPILE_ALL_ATTAINABLE)
  target_link_libraries(neo_dct_tests PRIVATE neo_dct hwy Threads::Threads)
  add_test(NAME dct.numerical COMMAND neo_dct_tests)
  set_tests_properties(dct.numerical PROPERTIES TIMEOUT 60)
  add_executable(neo_dct_c_test tests/c_api.c)
  target_link_libraries(neo_dct_c_test PRIVATE neo_dct)
  set_target_properties(neo_dct_c_test PROPERTIES LINKER_LANGUAGE CXX)
  add_test(NAME dct.c_api COMMAND neo_dct_c_test)
endif()
